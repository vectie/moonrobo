cmake_minimum_required(VERSION 3.16)
project(moonrobo_e1_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(MOONROBO_E1_STATUS_TESTS_ONLY "Build device-free CDR and deadline tests without the vendor SDK" OFF)
if(MOONROBO_E1_STATUS_TESTS_ONLY)
  enable_testing()
  add_executable(e1_status_cdr_test tests/status_cdr_test.cpp)
  add_test(NAME e1_status_cdr COMMAND e1_status_cdr_test)
  add_executable(e1_low_guard_test lowcontrol_candidate/guard_test.cpp)
  add_test(NAME e1_low_guard COMMAND e1_low_guard_test)
  return()
endif()
set(MOONROBO_E1_SDK_API "legacy" CACHE STRING "SDK adapter: legacy getters or 3.1 callbacks")
set_property(CACHE MOONROBO_E1_SDK_API PROPERTY STRINGS legacy 3.1)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../sdk" CACHE PATH "SDK E1 checkout")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(moonrobo_sdk_arch aarch64)
else()
  set(moonrobo_sdk_arch x86_64)
endif()
set(SDK_LIBRARY_DIR "${SDK_ROOT}/lib/${moonrobo_sdk_arch}" CACHE PATH "SDK target-architecture libraries")
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(moonrobo_adapter_targets moonrobo_e1)
if(MOONROBO_E1_SDK_API STREQUAL "3.1")
  set(SDK_DDS_LIBRARY_DIR "${SDK_ROOT}/dds/${moonrobo_sdk_arch}/lib" CACHE PATH "SDK3.1 CycloneDDS libraries")
  find_library(moonrobo_sdk_library NAMES noetix_e1_sdk PATHS "${SDK_LIBRARY_DIR}" NO_DEFAULT_PATH)
  find_library(moonrobo_dds_library NAMES ddsc PATHS "${SDK_DDS_LIBRARY_DIR}" NO_DEFAULT_PATH)
  find_library(moonrobo_ddsxx_library NAMES ddscxx PATHS "${SDK_DDS_LIBRARY_DIR}" NO_DEFAULT_PATH)
  if(NOT moonrobo_sdk_library OR NOT moonrobo_dds_library OR NOT moonrobo_ddsxx_library)
    message(FATAL_ERROR "SDK3.1 requires noetix_e1_sdk in SDK_LIBRARY_DIR and ddsc/ddscxx in SDK_DDS_LIBRARY_DIR")
  endif()
  add_library(moonrobo_e1 SHARED sdk_adapter_v31.cpp)
  add_executable(moonrobo_e1_low_worker_v1 ../sdk_e1_low/worker.cpp ../sdk_e1_low/sdk_adapter.cpp)
  add_library(moonrobo_e1_low SHARED ../sdk_e1_low/supervisor.cpp)
  add_dependencies(moonrobo_e1_low moonrobo_e1_low_worker_v1)
  target_link_libraries(moonrobo_e1_low PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  add_library(moonrobo_e1_status SHARED status_reader.cpp)
  target_include_directories(moonrobo_e1_status PRIVATE "${SDK_ROOT}/dds/${moonrobo_sdk_arch}/include")
  target_link_libraries(moonrobo_e1_status PRIVATE Threads::Threads "${moonrobo_dds_library}")
  if(BUILD_TESTING)
    enable_testing()
    add_executable(e1_status_cdr_test tests/status_cdr_test.cpp)
    add_test(NAME e1_status_cdr COMMAND e1_status_cdr_test)
  endif()
  list(APPEND moonrobo_adapter_targets moonrobo_e1_low_worker_v1)
  target_link_libraries(moonrobo_e1 PRIVATE Threads::Threads "${moonrobo_sdk_library}" "${moonrobo_dds_library}" "${moonrobo_ddsxx_library}")
  target_link_libraries(moonrobo_e1_low_worker_v1 PRIVATE Threads::Threads "${moonrobo_sdk_library}" "${moonrobo_dds_library}" "${moonrobo_ddsxx_library}")
elseif(MOONROBO_E1_SDK_API STREQUAL "legacy")
  add_library(moonrobo_e1 SHARED sdk_adapter.cpp "${SDK_ROOT}/src/highcontroller.cpp")
  target_link_directories(moonrobo_e1 PRIVATE "${SDK_LIBRARY_DIR}")
  target_link_libraries(moonrobo_e1 PRIVATE Threads::Threads driver ddsc ddscxx yaml-cpp)
else()
  message(FATAL_ERROR "Set MOONROBO_E1_SDK_API to legacy or 3.1")
endif()
foreach(adapter IN LISTS moonrobo_adapter_targets)
target_include_directories(${adapter} PRIVATE "${SDK_ROOT}/include" "${SDK_ROOT}/include/idl" "${SDK_ROOT}/include/ddscxx" ${EIGEN3_INCLUDE_DIRS})
# SDK3.1 common.h spells its include <eigen3/Eigen/Dense>, while other
# headers use <Eigen/Dense>. Admit both forms for a non-system Eigen prefix.
foreach(eigen_include IN LISTS EIGEN3_INCLUDE_DIRS)
  get_filename_component(eigen_parent "${eigen_include}" DIRECTORY)
  target_include_directories(${adapter} PRIVATE "${eigen_parent}")
endforeach()
target_compile_definitions(${adapter} PRIVATE SDK_ROOT="${SDK_ROOT}")
endforeach()
