cmake_minimum_required(VERSION 3.24)
project(moonrobo_aligned_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MUJOCO_INCLUDE_DIR "" CACHE PATH "Directory containing mujoco/mujoco.h")
set(MUJOCO_LIBRARY "" CACHE FILEPATH "Exact matching native MuJoCo library")
if(NOT EXISTS "${MUJOCO_INCLUDE_DIR}/mujoco/mujoco.h" OR NOT EXISTS "${MUJOCO_LIBRARY}")
  message(FATAL_ERROR "Set MUJOCO_INCLUDE_DIR and MUJOCO_LIBRARY; no simulator substitutes or downloads are used")
endif()
option(MOONROBO_CUDA_STATE "Compile genuine CUDA device-state/reset/target kernels (not a Newton solver)" OFF)
if(MOONROBO_CUDA_STATE)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(moonrobo_aligned_cuda_state SHARED cuda_state.cu)
  target_link_libraries(moonrobo_aligned_cuda_state PRIVATE CUDA::cudart)
  set_target_properties(moonrobo_aligned_cuda_state PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
endif()
option(MOONROBO_WARP_APIC "Compile actual Warp 1.17 CUDA APIC physics replay" OFF)
set(WARP_NATIVE_INCLUDE_DIR "" CACHE PATH "Pinned Warp 1.17 warp/native headers")
set(WARP_NATIVE_LIBRARY "" CACHE FILEPATH "Pinned CUDA-enabled Warp 1.17 native library")
if(MOONROBO_WARP_APIC)
  find_package(CUDAToolkit REQUIRED)
  if(NOT EXISTS "${WARP_NATIVE_INCLUDE_DIR}/apic.h" OR NOT EXISTS "${WARP_NATIVE_LIBRARY}")
    message(FATAL_ERROR "APIC replay requires real matching Warp 1.17 headers/library and CUDA SDK")
  endif()
  add_library(moonrobo_aligned_apic SHARED aot_solver.cpp)
  target_include_directories(moonrobo_aligned_apic PRIVATE "${WARP_NATIVE_INCLUDE_DIR}")
  target_link_libraries(moonrobo_aligned_apic PRIVATE CUDA::cuda_driver "${WARP_NATIVE_LIBRARY}")
endif()
add_library(moonrobo_aligned_usd SHARED usd_converter.cpp)
target_include_directories(moonrobo_aligned_usd PRIVATE "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(moonrobo_aligned_usd PRIVATE "${MUJOCO_LIBRARY}")
set_target_properties(moonrobo_aligned_usd PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
option(MOONROBO_OPENUSD_VALIDATOR "Compile a genuine OpenUSD stage validator (never execute automatically)" OFF)
if(MOONROBO_OPENUSD_VALIDATOR)
  find_package(pxr CONFIG REQUIRED)
  add_executable(moonrobo_validate_usd validate_usd.cpp)
  target_link_libraries(moonrobo_validate_usd PRIVATE usd usdGeom usdPhysics tf)
endif()
