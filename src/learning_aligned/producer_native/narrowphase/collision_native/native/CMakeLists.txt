cmake_minimum_required(VERSION 3.24)
project(moonrobo_collision_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
set(MUJOCO_INCLUDE_DIR "" CACHE PATH "Genuine MuJoCo 3.3.7 include directory")
if(NOT EXISTS "${MUJOCO_INCLUDE_DIR}/mujoco/mujoco.h")
  message(FATAL_ERROR "Explicit genuine MuJoCo headers required; no fallback or download")
endif()
add_library(moonrobo_collision_native SHARED collision.cpp)
target_include_directories(moonrobo_collision_native PRIVATE "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(moonrobo_collision_native PRIVATE ${CMAKE_DL_LIBS})
