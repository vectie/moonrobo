cmake_minimum_required(VERSION 3.20)
project(moonrobo_warp_dispatch LANGUAGES CXX)
set(WARP_NATIVE_INCLUDE_DIR "" CACHE PATH "Native headers from NVIDIA Warp v1.17.0")
if(NOT EXISTS "${WARP_NATIVE_INCLUDE_DIR}/warp.h" OR NOT EXISTS "${WARP_NATIVE_INCLUDE_DIR}/apic_types.h")
  message(FATAL_ERROR "Set WARP_NATIVE_INCLUDE_DIR to the actual pinned Warp native headers")
endif()
add_library(moonrobo_warp_dispatch SHARED dispatch.cpp)
target_include_directories(moonrobo_warp_dispatch SYSTEM PRIVATE "${WARP_NATIVE_INCLUDE_DIR}")
target_compile_features(moonrobo_warp_dispatch PRIVATE cxx_std_17)
target_link_libraries(moonrobo_warp_dispatch PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(moonrobo_warp_dispatch PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
