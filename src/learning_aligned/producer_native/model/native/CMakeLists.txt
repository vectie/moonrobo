cmake_minimum_required(VERSION 3.24)
project(moonrobo_producer_native_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
option(MOONROBO_PREPARE_MUJOCO "Build real MuJoCo model descriptor plugin" ON)
option(MOONROBO_PREPARE_OPENUSD "Build real OpenUSD metadata/variant/dependency plugin" OFF)
set(MUJOCO_INCLUDE_DIR "" CACHE PATH "Exact MuJoCo include directory; runtime version must match")
if(MOONROBO_PREPARE_MUJOCO)
  if(NOT EXISTS "${MUJOCO_INCLUDE_DIR}/mujoco/mujoco.h")
    message(FATAL_ERROR "Explicit real MUJOCO_INCLUDE_DIR required; no fallback headers/downloads")
  endif()
  add_library(moonrobo_prepared_model SHARED mujoco_model.cpp)
  target_include_directories(moonrobo_prepared_model PRIVATE "${MUJOCO_INCLUDE_DIR}")
  target_link_libraries(moonrobo_prepared_model PRIVATE ${CMAKE_DL_LIBS})
endif()
if(MOONROBO_PREPARE_OPENUSD)
  find_package(pxr CONFIG REQUIRED)
  if(PXR_ENABLE_PYTHON_SUPPORT OR PXR_PYTHON_SUPPORT_ENABLED)
    message(FATAL_ERROR "OpenUSD must be configured without Python support")
  endif()
  add_library(moonrobo_prepared_usd SHARED usd_metadata.cpp)
  target_link_libraries(moonrobo_prepared_usd PRIVATE usd usdGeom usdPhysics usdUtils usdShade sdf tf pcp)
endif()
