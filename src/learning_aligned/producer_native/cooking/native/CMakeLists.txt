cmake_minimum_required(VERSION 3.24)
project(moonrobo_host_mesh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
set(WARP_NATIVE_INCLUDE_DIR "" CACHE PATH "Genuine Warp 1.17.0 warp/native directory")
if(NOT EXISTS "${WARP_NATIVE_INCLUDE_DIR}/warp.h" OR NOT EXISTS "${WARP_NATIVE_INCLUDE_DIR}/mesh.h")
  message(FATAL_ERROR "Explicit genuine Warp native headers required; no fallback/download")
endif()
add_library(moonrobo_host_mesh SHARED host_mesh.cpp)
if(NOT WIN32)
  target_sources(moonrobo_host_mesh PRIVATE field.cpp)
endif()
target_include_directories(moonrobo_host_mesh PRIVATE "${WARP_NATIVE_INCLUDE_DIR}")
target_compile_definitions(moonrobo_host_mesh PRIVATE WP_ENABLE_CUDA=0)
target_link_libraries(moonrobo_host_mesh PRIVATE ${CMAKE_DL_LIBS})
option(MOONROBO_COOK_QHULL "Build optional genuine reentrant Qhull adapter" OFF)
if(MOONROBO_COOK_QHULL)
  set(QHULL_INCLUDE_DIR "" CACHE PATH "Qhull 2020.2 src directory containing libqhull_r")
  if(NOT EXISTS "${QHULL_INCLUDE_DIR}/libqhull_r/qhull_ra.h")
    message(FATAL_ERROR "Explicit genuine Qhull 2020.2 headers required")
  endif()
  if(WIN32)
    message(FATAL_ERROR "Qhull dynamic adapter currently requires POSIX dlopen")
  endif()
  add_library(moonrobo_convex_hull SHARED hull.cpp)
  target_include_directories(moonrobo_convex_hull PRIVATE "${QHULL_INCLUDE_DIR}")
  target_link_libraries(moonrobo_convex_hull PRIVATE ${CMAKE_DL_LIBS})
endif()
