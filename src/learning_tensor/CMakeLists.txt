cmake_minimum_required(VERSION 3.18)
project(moonrobo_learning_tensor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Torch REQUIRED)
add_library(moonrobo_learning_tensor SHARED adapter.cpp)
target_include_directories(moonrobo_learning_tensor PRIVATE ${TORCH_INCLUDE_DIRS})
target_link_libraries(moonrobo_learning_tensor PRIVATE ${TORCH_LIBRARIES} ${CMAKE_DL_LIBS})
option(MR_ENABLE_CUDA_GRAPH "Build the optional PyTorch 2.11 CUDA graph extension" OFF)
option(MR_ENABLE_GEOMETRY_FUSION "Build bounded geometry CUDA kernels" OFF)
if(MR_ENABLE_GEOMETRY_FUSION)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set_property(TARGET moonrobo_learning_tensor PROPERTY CUDA_STANDARD 17)
  target_sources(moonrobo_learning_tensor PRIVATE compact_heading.cu fused_rotation.cu geometry_tail.cu geometry_vjp.cu)
  target_compile_definitions(moonrobo_learning_tensor PRIVATE MR_ENABLE_GEOMETRY_FUSION=1)
  target_compile_options(moonrobo_learning_tensor PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>)
  target_link_libraries(moonrobo_learning_tensor PRIVATE CUDA::cudart)
endif()
option(MR_ENABLE_OPERATOR_DIAGNOSTICS "Build independent ATen/autograd NVTX diagnostics (requires CUDA and NVTX3 headers)" OFF)
if(MR_ENABLE_OPERATOR_DIAGNOSTICS)
  target_compile_definitions(moonrobo_learning_tensor PRIVATE MR_ENABLE_OPERATOR_DIAGNOSTICS=1 MR_ENABLE_NVTX3=1)
endif()
if(MR_ENABLE_CUDA_GRAPH)
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(moonrobo_learning_tensor PRIVATE MR_ENABLE_CUDA_GRAPH=1)
  target_link_libraries(moonrobo_learning_tensor PRIVATE CUDA::cudart)
endif()
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
